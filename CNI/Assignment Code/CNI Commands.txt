9. ACCESS CONTROL LISTS (ACL) - GENERAL SYNTAX & CONFIGURATION TASKS
--------------------------------------------------------------------------------
! ==========================================
! ACL GENERAL SYNTAX REFERENCE
! ==========================================
! 1. Standard Numbered ACL Syntax (ID Range: 1 - 99):
!    Router(config)# access-list <1-99> {permit|deny} <source-ip> [wildcard-mask]
!    Router(config)# access-list <1-99> {permit|deny} host <source-ip>
!    Router(config)# access-list <1-99> {permit|deny} any
!
! 2. Extended Numbered ACL Syntax (ID Range: 100 - 199):
!    Router(config)# access-list <100-199> {permit|deny} <protocol> <source-ip> <source-wildcard> <dest-ip> <dest-wildcard> [eq port]
!    Note: <protocol> can be ip, tcp, udp, icmp, etc.
!    Note: Common port numbers / keywords: 23 (telnet), 80 (www/http), 21 (ftp), 53 (domain/dns)
!
! 3. Named ACL Syntax (Standard or Extended):
!    Router(config)# ip access-list {standard | extended} <acl-name>
!    Router(config-std-nacl)# {permit | deny} <source-ip> [wildcard-mask]
!    Router(config-ext-nacl)# {permit | deny} <protocol> <source> <source-wildcard> <destination> <dest-wildcard> [eq port]
!
! 4. Applying ACL to an Interface:
!    Router(config-if)# ip access-group <acl-number-or-name> {in | out}
!
! 5. Verification & Management Commands:
!    Router# show access-lists
!    Router# show ip interface <interface-id>

! ==========================================
! PRACTICAL LAB CONFIGURATION TASKS
! ==========================================

! Task 1: Standard ACL - Block a particular host from accessing all connected 
!         devices on the second router.
! Note: Standard ACLs filter based on source IP and are placed close to destination.
Router2> enable
Router2# configure terminal
Router2(config)# access-list 10 deny host 192.168.10.11
Router2(config)# access-list 10 permit any
Router2(config)# interface FastEthernet 0/0
Router2(config-if)# ip access-group 10 out
Router2(config-if)# exit
Router2(config)# do write memory

! Task 2: Extended ACL - Block one entire network from accessing another network.
! Note: Extended ACLs filter source to destination and are placed close to source.
Router1> enable
Router1# configure terminal
Router1(config)# access-list 100 deny ip 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255
Router1(config)# access-list 100 permit ip any any
Router1(config)# interface FastEthernet 0/0
Router1(config-if)# ip access-group 100 in
Router1(config-if)# exit
Router1(config)# do write memory

! Task 3: Extended ACL - Block Telnet (TCP 23) and HTTP (TCP 80) between two networks.
Router1> enable
Router1# configure terminal
Router1(config)# access-list 101 deny tcp 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255 eq 23
Router1(config)# access-list 101 deny tcp 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255 eq 80
Router1(config)# access-list 101 permit ip any any
Router1(config)# interface FastEthernet 0/0
Router1(config-if)# ip access-group 101 in
Router1(config-if)# exit
Router1(config)# do write memory

Router1(config-if)# ip nat inside
Router1(config-if)# exit
Router1(config)# interface Serial 0/0/0
Router1(config-if)# ip nat outside
